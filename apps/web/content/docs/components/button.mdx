## Installation

```npm
npx shadcn@latest add button -c apps/web
```

## Usage

```tsx title="components/example.tsx"
import { Button } from "@workspace/ui/components/button"

export function Example() {
  return <Button variant="outline">Button</Button>
}
```

## Variants

Use the `variant` prop: `default`, `secondary`, `outline`, `ghost`, `destructive` and `link`.

### Sizes

Use the `size` prop: `default`, `xs`, `sm`, `lg` and the `icon` sizes.

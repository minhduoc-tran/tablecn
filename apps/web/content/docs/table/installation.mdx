## Requirements

- React 18 or 19
- A project set up with [shadcn/ui](https://ui.shadcn.com/docs/installation) and Tailwind CSS v4
- TanStack Table v9.2+ (installed for you)
- For the filter: the [filter packages](/docs/filter/installation)

## 1. Add the UI

The `data-table` block exists in three versions, one per primitive library. Pick the one matching the `style` in your `components.json`:

| `style` in components.json | Primitives | Registry URL |
| --- | --- | --- |
| `radix-*` (e.g. `radix-nova`) | Radix UI | `%SITE_URL%/r/radix/data-table.json` |
| `base-*` (e.g. `base-nova`) | Base UI | `%SITE_URL%/r/base/data-table.json` |
| `aria-*` (e.g. `aria-nova`) | React Aria | `%SITE_URL%/r/aria/data-table.json` |

```npm
npx shadcn@latest add %SITE_URL%/r/radix/data-table.json
```

The CLI:

- copies the table, its header, pagination, the **Columns** menu, the toolbar and their helpers into `components/data-table/` (or wherever your `components` alias points), rewriting imports to your aliases;
- installs `@querycn/table-react`, `@querycn/filter-react`, `@tanstack/react-table`, `@tanstack/react-virtual`, `@dnd-kit/core`, `@dnd-kit/sortable` and `lucide-react`;
- adds the shadcn components it builds on if you don't have them: `button`, `checkbox`, `dropdown-menu`, `popover`, `select`, `separator`, `skeleton`, `table` and `tooltip`.

For the filter UI in the toolbar, add the [`filter-builder`](/docs/filter/installation#1-add-the-ui) block too.

<Callout title="Only want the logic?">
  Skip this step and install the package yourself: `npm install @querycn/table-react @querycn/filter-react @tanstack/react-table`. `useDataTable` returns a TanStack table, so any table UI can render it.
</Callout>

## 2. Define the columns

Columns are [TanStack column definitions](https://tanstack.com/table/latest/docs/guide/column-defs). Build them with `createDataTableColumnHelper`, which types the [`meta` keys](/docs/table/columns#meta) the table reads:

```tsx title="components/orders/order-columns.tsx"
"use client"

import { createDataTableColumnHelper } from "@querycn/table-react"

import { createSelectionColumn } from "@/components/data-table/data-table-selection-column"

export interface Order {
  id: string
  customer: string
  status: "paid" | "pending"
  amount: number
}

const helper = createDataTableColumnHelper<Order>()

export const orderColumns = [
  createSelectionColumn<Order>(),
  helper.accessor("customer", { header: "Customer", size: 200 }),
  helper.accessor("status", { header: "Status" }),
  helper.accessor("amount", { header: "Amount", sortDescFirst: true }),
]
```

Define them outside the component, or in `useMemo`: TanStack Table expects `columns` and `data` to keep their identity between renders.

## 3. Render the table in a FilterProvider

`useDataTable` reads and writes the URL through an adapter. Under a `FilterProvider` it uses the provider's adapter, so the filter, sort and page share one URL. The provider's `onApply` returns `resetPagePatch()`, so a new filter goes back to page 1 in the same navigation:

```tsx title="components/orders/orders-table.tsx"
"use client"

import { useDataTable } from "@querycn/table-react"

import { DataTable } from "@/components/data-table/data-table"
import { DataTablePagination } from "@/components/data-table/data-table-pagination"
import { DataTableToolbar } from "@/components/data-table/data-table-toolbar"
import { FilterBuilder } from "@/components/filter/filter-builder"
import { FilterChips } from "@/components/filter/filter-chips"
import { orderColumns, type Order } from "./order-columns"

export function OrdersTable({ orders }: { orders: Order[] }) {
  const table = useDataTable({
    data: orders,
    columns: orderColumns,
    getRowId: (order) => order.id,
    storageKey: "orders-table",
  })

  return (
    <div className="flex flex-col gap-3">
      <DataTableToolbar table={table}>
        <FilterBuilder />
        <FilterChips />
      </DataTableToolbar>
      <DataTable table={table} className="max-h-[600px]" />
      <DataTablePagination table={table} />
    </div>
  )
}
```

`storageKey` saves the column layout in `localStorage`. Now wrap it in the provider for your router:

<Tabs items={["Next.js", "Vite / plain React", "react-router"]}>
<Tab value="Next.js">

```tsx title="app/orders/orders-view.tsx"
"use client"

import { NextFilterProvider } from "@querycn/filter-next"
import { resetPagePatch } from "@querycn/table-react"

import { orderFields } from "@/lib/order-fields"
import { OrdersTable } from "@/components/orders/orders-table"

export function OrdersView({ orders }: { orders: Order[] }) {
  return (
    <NextFilterProvider
      fields={orderFields}
      shallow
      onApply={() => resetPagePatch()}
    >
      <OrdersTable orders={orders} />
    </NextFilterProvider>
  )
}
```

`shallow` updates the URL without a server request, which is what you want when every row is already in the browser. When a server component reads `searchParams` to fetch one page, leave it off; see [Server data](/docs/table/server-data#rendering-on-the-server).

`useSearchParams` needs a `<Suspense>` boundary on prerendered pages; see [Next.js](/docs/filter/nextjs) for the details.

```tsx title="src/orders-page.tsx"
import { FilterProvider, useBrowserUrlAdapter } from "@querycn/filter-react"
import { resetPagePatch } from "@querycn/table-react"

import { orderFields } from "@/lib/order-fields"
import { OrdersTable } from "@/components/orders/orders-table"

export function OrdersPage({ orders }: { orders: Order[] }) {
  const adapter = useBrowserUrlAdapter()

  return (
    <FilterProvider
      fields={orderFields}
      adapter={adapter}
      onApply={() => resetPagePatch()}
    >
      <OrdersTable orders={orders} />
    </FilterProvider>
  )
}
```

<Tab value="react-router">

```tsx title="src/routes/orders.tsx"
import { FilterProvider } from "@querycn/filter-react"
import { useReactRouterAdapter } from "@querycn/filter-react/react-router"
import { resetPagePatch } from "@querycn/table-react"

import { orderFields } from "@/lib/order-fields"
import { OrdersTable } from "@/components/orders/orders-table"

export function OrdersRoute() {
  const adapter = useReactRouterAdapter()
  const orders = useOrders() // your data

  return (
    <FilterProvider
      fields={orderFields}
      adapter={adapter}
      onApply={() => resetPagePatch()}
    >
      <OrdersTable orders={orders} />
    </FilterProvider>
  )
}
```

The fields are the filter's; see [Fields](/docs/filter/fields). Their names are matched against the rows' keys, so the rows are filtered in the browser without more setup. For nested values (`customer.name`), pass [`getFilterValue`](/docs/table/server-data#client-mode).

### Without the filter

The table doesn't need a `FilterProvider`. Pass an adapter to `useDataTable` instead, and the clear filters button stays hidden:

```tsx title="Without a filter"
import { useNextAdapter } from "@querycn/filter-next"

const adapter = useNextAdapter({ shallow: true })
const table = useDataTable({ data, columns, getRowId, adapter })
```

With neither, sorting and pages live in memory and the URL is left alone.
